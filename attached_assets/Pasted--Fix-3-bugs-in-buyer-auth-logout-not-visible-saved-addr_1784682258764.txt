// Pre-fill address whenever buyerProfile becomes available
useEffect(() => {
  if (!buyerProfile) return;
  setForm(prev => {
    // Only fill empty fields — don't overwrite what buyer already typed
    return {
      name: prev.name || buyerProfile.full_name || '',
      phone: prev.phone || buyerProfile.phone || '',
      address: prev.address || buyerProfile.default_address || '',
      city: prev.city || buyerProfile.default_city || '',
      state: prev.state || buyerProfile.default_state || '',

    };
  });
}, [buyerProfile]); // runs every time profile loads or changes

function BuyerProfileSheet({ onClose }: { onClose: () => void }) {
  const { buyerSession, buyerProfile, refreshProfile } = useBuyerAuth();
  const [form, setForm] = useState({
    full_name: buyerProfile?.full_name ?? '',
    phone: buyerProfile?.phone ?? '',
    default_address: buyerProfile?.default_address ?? '',
    default_city: buyerProfile?.default_city ?? '',
    default_state: buyerProfile?.default_state ?? '',
    default_pincode: buyerProfile?.default_pincode ?? '',
  });

  // Sync form when profile loads (handles late-loading case)

    if (!buyerProfile) return;
    setForm({
      full_name: buyerProfile.full_name ?? '',
      phone: buyerProfile.phone ?? '',
      default_address: buyerProfile.default_address ?? '',
      default_city: buyerProfile.default_city ?? '',
      default_state: buyerProfile.default_state ?? '',
